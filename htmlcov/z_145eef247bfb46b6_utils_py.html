19def get_transactions_from_json(path: str) -> List[Dict]: 

20 """ 

21 Load a list of transactions from a JSON file. 

22 

23 :param path: Path to the JSON file containing transaction data. 

24 :return: A list of dictionaries representing transactions. Returns an empty list if the file 

25 is not found, cannot be decoded, or does not contain a list. 

26 """ 

27 utils_logger.debug(f"Attempting to load transactions from {path}") 

28 try: 

29 with open(path, "r", encoding="utf-8") as f: 

30 data = json.load(f) 

31 if isinstance(data, list): 

32 utils_logger.info(f"Successfully loaded {len(data)} transactions from {path}") 

33 return data 

34 else: 

35 utils_logger.warning(f"Invalid data format in {path}: expected list, got {type(data).__name__}") 

36 # Поставил уровень WARNING здесь и далее в этой функции, т.к. она обрабатывает 

37 # исключения "в тихую", не выбрасывая исключение, просто возвращает пустой список. 

38 # Считаю целесообразным выводить в лог предупреждение, а не ошибку. 

39 except FileNotFoundError: 

40 utils_logger.warning(f"File not found: {path}") 

41 except json.JSONDecodeError as e: 

42 utils_logger.warning(f"Failed to decode JSON from {path}: {e}") 

43 

44 utils_logger.debug(f"Returning empty transaction list for {path}") 

45 return []
